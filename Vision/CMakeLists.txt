cmake_minimum_required(VERSION 3.16)
project(rm_vision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RM_VISION_ENABLED "0" CACHE INTERNAL "Whether vision is compiled in" FORCE)
option(RM_ENABLE_VISION "Compile traditional green light vision" ON)
if(NOT RM_ENABLE_VISION)
    return()
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d video imgcodecs)
find_package(Threads REQUIRED)
set(RM_VISION_ENABLED "1" CACHE INTERNAL "Whether vision is compiled in" FORCE)
add_library(vision_core STATIC VisionPipeline.cpp TargetTracker.cpp SerialOutput.cpp)
target_include_directories(vision_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vision_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
target_compile_options(vision_core PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(vision_geometry_test GeometryTest.cpp)
target_link_libraries(vision_geometry_test PRIVATE vision_core)
add_test(NAME vision_geometry_test COMMAND vision_geometry_test "${CMAKE_SOURCE_DIR}/camera_intrinsics.yaml")
add_executable(vision_serial_test SerialTest.cpp)
target_link_libraries(vision_serial_test PRIVATE vision_core)
add_test(NAME vision_serial_test COMMAND vision_serial_test)
add_executable(vision_image_test ImageTest.cpp)
target_link_libraries(vision_image_test PRIVATE vision_core)
